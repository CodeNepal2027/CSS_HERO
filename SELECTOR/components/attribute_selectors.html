🟣 BOX-1 → presence [ attr ][href] { background: #ef4444; } /* ✓ matches any element that HAS the href attribute ✗ does NOT match elements without href */

🟣 BOX-3 → whole-word match [ attr~="value" ][class~="active"] { background: #f59e0b; } /* ✓ matches when "active" is a WHOLE WORD in a space-separated list ✓ class="btn active" · ✓ class="active" ✗ class="btn inactive" (inactive ≠ active) */

🟣 BOX-4 → starts-with [ attr^="value" ][href^="https"] { background: #8b5cf6; } /* ✓ matches href that STARTS with "https" ✗ does NOT match "http://..." */

🟣 BOX-5 → ends-with [ attr$="value" ][href$=".pdf"] { background: #06b6d4; } /* ✓ matches href that ENDS with ".pdf" ✗ does NOT match ".docx" — great for file-type icons */
href="/docs/intro.html" href="/blog/post.html"
🟣 BOX-6 → contains [ attr*="value" ][href*="docs"] { background: #ec4899; } /* ✓ matches href that CONTAINS "docs" ANYWHERE ✗ does NOT match "/blog/..." */
